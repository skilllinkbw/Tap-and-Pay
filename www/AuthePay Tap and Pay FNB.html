📷 CardScanner.kt

// CardScanner.kt - Wrapper for TapCardScannerKit package com.authepay.softpos import android.app.Activity import com.tap.cardscanner.CardScanner import com.tap.cardscanner.models.CardDetails class CardScanner { suspend fun scanCard(activity: Activity): CardData { return suspendCoroutine { continuation -> val scanner = CardScanner() scanner.startScan(activity) { cardDetails: CardDetails? -> if (cardDetails != null) { val cardData = CardData( cardNumber = cardDetails.cardNumber, expiryMonth = cardDetails.expiryMonth, expiryYear = cardDetails.expiryYear, cvv = cardDetails.cvv ) // Mask PAN in logs val maskedPan = cardDetails.cardNumber.replace( Regex("\\d(?=\\d{4})"), "*" ) android.util.Log.d("CardScanner", "Scanned: $maskedPan") continuation.resume(cardData) } else { continuation.resumeWithException( Exception("Card scan failed") ) } } } } } data class CardData( val cardNumber: String, val expiryMonth: String, val expiryYear: String, val cvv: String )

📡 NfcReader.kt

// NfcReader.kt - Wrapper for tap-to-pay-android package com.authepay.softpos import android.app.Activity import android.nfc.NfcAdapter import com.pedro.kt.tapToPay.NfcTapToPay import com.pedro.kt.tapToPay.models.PaymentData class NfcReader { suspend fun readNfcCard(activity: Activity): TokenizedCardData { return suspendCoroutine { continuation -> val nfcAdapter = NfcAdapter.getDefaultAdapter(activity) if (nfcAdapter == null) { continuation.resumeWithException( Exception("NFC not supported") ) return@suspendCoroutine } val tapToPay = NfcTapToPay(activity) tapToPay.startReading( onSuccess = { paymentData: PaymentData -> val tokenData = TokenizedCardData( token = paymentData.token, lastFour = paymentData.lastFour, expiry = paymentData.expiry ) android.util.Log.d("NfcReader", "NFC read successful") continuation.resume(tokenData) }, onError = { error: Throwable -> continuation.resumeWithException(error) } ) } } } data class TokenizedCardData( val token: String, val lastFour: String, val expiry: String )
